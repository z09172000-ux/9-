function doGet(e) { var sheet = getSheet(); return respond(readAll(sheet)); } function doPost(e) { var sheet = getSheet(); var payload = JSON.parse(e.postData.contents); var action = payload.action; if (action === 'list') { return respond(readAll(sheet)); } if (action === 'upsert') { var entry = payload.entry; var data = readAll(sheet); var idx = -1; for (var i = 0; i < data.length; i++) { if (String(data[i].id) === String(entry.id)) { idx = i; break; } } if (idx >= 0) { writeRow(sheet, idx + 2, entry); } else { sheet.appendRow([entry.id, entry.name, entry.idnum, entry.phone, entry.birthdate, entry.age, entry.edu, entry.createdAt]); } return respond(readAll(sheet)); } if (action === 'delete') { var id = payload.id; var data = readAll(sheet); for (var j = 0; j < data.length; j++) { if (String(data[j].id) === String(id)) { sheet.deleteRow(j + 2); break; } } return respond(readAll(sheet)); } if (action === 'clear') { var lastRow = sheet.getLastRow(); if (lastRow > 1) { sheet.deleteRows(2, lastRow - 1); } return respond([]); } return respond({ error: 'unknown action' }); } function getSheet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('Data'); var header = ['id', 'name', 'idnum', 'phone', 'birthdate', 'age', 'edu', 'createdAt']; if (!sheet) { sheet = ss.insertSheet('Data'); sheet.appendRow(header); return sheet; } // 若表頭跟目前欄位不一致（例如沿用舊版試算表），自動重設表頭 var headerRow = sheet.getRange(1, 1, 1, Math.max(sheet.getLastColumn(), 1)).getValues()[0]; if (headerRow.indexOf('birthdate') === -1) { sheet.getRange(1, 1, 1, header.length).setValues([header]); } return sheet; } function readAll(sheet) { var values = sheet.getDataRange().getValues(); var rows = values.slice(1); var result = []; for (var i = 0; i < rows.length; i++) { var r = rows[i]; if (!r[0]) continue; result.push({ id: String(r[0]), name: r[1], idnum: r[2], phone: r[3], birthdate: r[4], age: r[5], edu: r[6], createdAt: r[7] }); } return result; } function writeRow(sheet, rowIndex, entry) { sheet.getRange(rowIndex, 1, 1, 8).setValues([[entry.id, entry.name, entry.idnum, entry.phone, entry.birthdate, entry.age, entry.edu, entry.createdAt]]); } function respond(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }
